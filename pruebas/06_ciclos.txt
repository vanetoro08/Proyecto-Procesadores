~ catalizador (mientras): cuenta de 0 a 2
(elemento i 0)
(catalizador (bajo i 3)
  (emite i)
  (enlace i (+ i 1)))

~ cristaliza (para): el limite final es inclusivo, [k 1 5] va de 1 a 5
(elemento suma 0)
(cristaliza [k 1 5]
  (emite k)
  (enlace suma (+ suma k)))
(emite (+ "Suma: " suma))

~ precipita (romper): sale del ciclo cuando j llega a 3
(elemento j 0)
(catalizador polar
  (valorar (isotopo j 3)
    (precipita))
  (emite j)
  (enlace j (+ j 1)))
(emite (+ "Salio en " j))
